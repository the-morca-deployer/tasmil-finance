# /traction Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A public, read-only traction dashboard at `tasmil.finance/traction` (foundation audience) backed by a new unauthenticated `GET /api/public/traction` endpoint.

**Architecture:** One new backend endpoint in the existing `public` NestJS module aggregates all-time KPIs (reusing `PublicService.getStats()`), a 90-day daily Volume/TVL series, a 90-day daily user-growth series, and an all-time tx-type breakdown into a single Redis-cached payload. The frontend adds a `(public)` route + `src/features/traction/` feature module consuming the Kubb-generated client, visually mirroring the admin analytics patterns.

**Tech Stack:** NestJS + Prisma raw SQL + Redis (backend); Next.js 16 App Router, TanStack Query via Kubb-generated hooks, recharts, shadcn-style shared UI, Jest + Testing Library (frontend).

**Spec:** `docs/superpowers/specs/2026-07-02-traction-page-design.md`

- [ ] **Step 5: Create the service**

Create `src/modules/public/traction.service.ts`:

```ts
import { Injectable, Logger, ServiceUnavailableException } from '@nestjs/common';
import { PrismaService } from '../../database/prisma.service';
import { RedisService } from '../../database/redis/redis.service';
import { PublicService } from './public.service';
import {
  TractionResponseDto,
  TractionTxTypeCountDto,
  TractionUserGrowthPointDto,
  TractionVolumeTvlPointDto,
} from './dto/traction-response.dto';

const TRACTION_CACHE_KEY = 'public:traction:v1';
const TRACTION_CACHE_TTL_SECONDS = 300;
const WINDOW_DAYS = 90;

@Injectable()
export class TractionService {
  private readonly logger = new Logger(TractionService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly redis: RedisService,
    private readonly publicService: PublicService,
  ) {}

  async getTraction(): Promise<TractionResponseDto> {
    const cached = await this.redis
      .getJson<TractionResponseDto>(TRACTION_CACHE_KEY)
      .catch(() => null);
    if (cached) return cached;

    try {
      const since = new Date(Date.now() - WINDOW_DAYS * 24 * 60 * 60 * 1000);
      const [stats, volumeTvl, userGrowth, txByType] = await Promise.all([
        this.publicService.getStats(),
        this.getVolumeTvlSeries(since),
        this.getUserGrowthSeries(since),
        this.getTxByType(),
      ]);

      const payload: TractionResponseDto = {
        summary: {
          totalTvlUsd: stats.totalTvlUsd,
          totalUsers: stats.totalUsers,
          avgApyPercent: stats.avgApyPercent,
          totalTransactions: stats.totalTransactions,
        },
        volumeTvl,
        userGrowth,
        txByType,
        updatedAt: new Date().toISOString(),
      };

      await this.redis
        .setJson(TRACTION_CACHE_KEY, payload, TRACTION_CACHE_TTL_SECONDS)
        .catch(() => null);

      return payload;
    } catch (error) {
      // No zeroed fallback (unlike getStats): a traction page silently showing
      // zeros to reviewers is worse than a visible error state.
      this.logger.error(
        `Failed to compute traction payload: ${error instanceof Error ? error.message : error}`,
      );
      throw new ServiceUnavailableException('Traction data temporarily unavailable');
    }
  }

  // Same aggregation as AdminService.getVolumeTvlSeries (admin.service.ts) with
  // granularity fixed to 'day'. Duplicated on purpose — extract a shared
  // analytics service only when a third consumer appears.
  private async getVolumeTvlSeries(since: Date): Promise<TractionVolumeTvlPointDto[]> {
    const buckets = await this.prisma.$queryRaw<
      Array<{ bucket: Date; volume_usd: string | null; net_usd: string | null }>
    >`
      SELECT
        DATE_TRUNC('day', created_at AT TIME ZONE 'UTC') AS bucket,
        SUM(CASE WHEN type IN ('DEPOSIT', 'WITHDRAW') THEN amount_usd ELSE 0 END) AS volume_usd,
        SUM(CASE WHEN type = 'DEPOSIT' THEN amount_usd
                 WHEN type = 'WITHDRAW' THEN -amount_usd
                 ELSE 0 END) AS net_usd
      FROM activities
      WHERE created_at >= ${since}
      GROUP BY bucket
      ORDER BY bucket ASC
    `;

    const priorNetRows = await this.prisma.$queryRaw<Array<{ net_usd: string | null }>>`
      SELECT SUM(
        CASE WHEN type = 'DEPOSIT' THEN amount_usd
             WHEN type = 'WITHDRAW' THEN -amount_usd
             ELSE 0 END
      ) AS net_usd
      FROM activities
      WHERE created_at < ${since}
    `;

    let cumulativeTvlUsd = Number(priorNetRows[0]?.net_usd ?? 0);
    return buckets.map((row) => {
      cumulativeTvlUsd += Number(row.net_usd ?? 0);
      return {
        date: row.bucket.toISOString().slice(0, 10),
        volumeUsd: Number(row.volume_usd ?? 0),
        cumulativeTvlUsd,
      };
    });
  }

  // Counts from the users table (not waitlist_entries like the admin dashboard)
  // so the curve's final value matches the totalUsers KPI beside it.
  private async getUserGrowthSeries(since: Date): Promise<TractionUserGrowthPointDto[]> {
    const rows = await this.prisma.$queryRaw<Array<{ date: string; count: number }>>`
      SELECT
        TO_CHAR(DATE_TRUNC('day', created_at AT TIME ZONE 'UTC'), 'YYYY-MM-DD') AS date,
        COUNT(*)::integer AS count
      FROM users
      WHERE created_at >= ${since}
      GROUP BY DATE_TRUNC('day', created_at AT TIME ZONE 'UTC')
      ORDER BY date ASC
    `;

    const priorCount = await this.prisma.user.count({ where: { createdAt: { lt: since } } });

    const countsByDate = new Map(rows.map((r) => [r.date, r.count]));
    const series: TractionUserGrowthPointDto[] = [];
    let cumulativeUsers = priorCount;
    for (let i = WINDOW_DAYS - 1; i >= 0; i--) {
      const d = new Date();
      d.setUTCDate(d.getUTCDate() - i);
      const date = d.toISOString().slice(0, 10);
      const newUsers = countsByDate.get(date) ?? 0;
      cumulativeUsers += newUsers;
      series.push({ date, newUsers, cumulativeUsers });
    }
    return series;
  }

  private async getTxByType(): Promise<TractionTxTypeCountDto[]> {
    const grouped = await this.prisma.activity.groupBy({
      by: ['type'],
      _count: { _all: true },
    });
    return grouped
      .map((g) => ({ type: g.type as string, count: g._count._all }))
      .sort((a, b) => b.count - a.count);
  }
}
```

```ts
import { Controller, Get } from '@nestjs/common';
import { ApiOperation, ApiResponse, ApiTags } from '@nestjs/swagger';

import { StatsResponseDto } from './dto/stats-response.dto';

  @ApiResponse({ status: 200, description: 'Traction payload', type: TractionResponseDto })
  @ApiResponse({ status: 503, description: 'Metrics temporarily unavailable' })
```

```ts
export function fmtUsdCompact(n: number): string {
  if (n >= 1_000_000) return `$${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000) return `$${(n / 1_000).toFixed(1)}k`;
  return `$${n.toFixed(0)}`;
}

export function fmtInt(n: number): string {
  return n.toLocaleString("en-US");
}

export function fmtPercent(n: number): string {
  return `${n.toFixed(2)}%`;
}
```

```tsx
import { Button } from "@/shared/ui/button";
import { Typography } from "@/shared/ui/typography";
import { useTraction } from "../hooks/use-traction";
import { KpiCards } from "./kpi-cards";
import { TxBreakdown } from "./tx-breakdown";
import { UserGrowthChart } from "./user-growth-chart";
import { VolumeTvlChart } from "./volume-tvl-chart";

export function TractionDashboard() {
  const { data, isLoading, isError, refetch } = useTraction();

  if (isError) {
    return (
      <div className="flex min-h-[60vh] flex-col items-center justify-center gap-4 px-4">
        <Typography variant="h2" className="text-xl">
          Data temporarily unavailable
        </Typography>
        <Typography variant="p" className="text-center text-muted-foreground text-sm">
          Live metrics could not be loaded. Please try again in a moment.
        </Typography>
        <Button variant="outline" onClick={() => refetch()}>
          Retry
        </Button>
      </div>
    );
  }

  return (
    <div className="mx-auto flex w-full max-w-5xl flex-col gap-6 px-4 py-10">
      <header className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex items-center gap-3">
          <Image src="/tasmil-logo.png" alt="Tasmil Finance" width={32} height={32} />
          <div>
            <h1 className="font-bold text-2xl">Tasmil Traction</h1>
            <p className="text-muted-foreground text-xs">
              Live growth metrics — Stellar mainnet
            </p>
          </div>
        </div>
        {data?.updatedAt && (
          <span className="rounded bg-green-500/10 px-2 py-1 font-semibold text-[11px] text-green-400">
            Live data — updated {new Date(data.updatedAt).toUTCString()}
          </span>
        )}
      </header>

      <KpiCards summary={data?.summary} isLoading={isLoading} />
      <VolumeTvlChart data={data?.volumeTvl} isLoading={isLoading} />
      <UserGrowthChart data={data?.userGrowth} isLoading={isLoading} />
      <TxBreakdown items={data?.txByType} isLoading={isLoading} />

      <footer className="mt-4 border-border border-t pt-4 text-center">
        <a
          href="https://tasmil.finance"
          className="text-muted-foreground text-xs hover:text-foreground"
        >
          tasmil.finance
        </a>
      </footer>
    </div>
  );
}
```
